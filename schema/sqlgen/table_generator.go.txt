package sqlgen

import (
	"fmt"
	"valsafe/schema/types"
	"strings"
)

func GenerateCreateTable(schema string, t *types.Table) string {

	var columns []string

	for name, col := range t.Columns {

		line := fmt.Sprintf("%s %s", name, col.Type)

		if col.NotNull {
			line += " NOT NULL"
		}

		if col.Unique {
			line += " UNIQUE"
		}

		if col.Default != "" {
			line += " DEFAULT " + col.Default
		}

		if col.PrimaryKey {
			line += " PRIMARY KEY"
		}

		columns = append(columns, line)
	}

	tableName := fmt.Sprintf("%s.%s", schema, t.Table)

	sql := fmt.Sprintf(
		"CREATE TABLE IF NOT EXISTS %s (%s);",
		tableName,
		strings.Join(columns, ","),
	)

	return sql
}
