# DEV
GOOS=linux GOARCH=amd64 go build -o bin/api-dev ./cmd/api

# PROD
GOOS=linux GOARCH=amd64 go build -tags=prod -o bin/api ./cmd/api


to set OS environmnet for linux
GOOS=linux GOARCH=amd64

to set OS environment for linux using powershell
$env:GOOS="linux"
$env:GOARCH="amd64"

or 

$env:GOOS="linux"; $env:GOARCH="amd64";

# Docker build (uses config-docker.go + redis_client_docker.go)
go build -tags=docker ./cmd/api ./cmd/worker ./cmd/schema

# Local dev build (uses config-dev.go + redis_client.go — unchanged behavior)
go build ./cmd/api

# Launch the stack
docker compose up --build -d
curl http://localhost:1040/health   # → {"status":"healthy"}
